數值型欄位：'ship_length', 'cargo_weight'

from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder

# 1. 區分數值型欄位和類別型欄位
num_cols = ['ship_length', 'cargo_weight']
cat_cols = ['ship_type', 'port']

# 2. 建立數值管道：SimpleImputer（mean）→ StandardScaler
num_pipeline = Pipeline([
    ('imputer', SimpleImputer(strategy='mean')),
    ('scaler', StandardScaler())
])

# 3. 建立類別管道：SimpleImputer（most_frequent）→ OneHotEncoder
cat_pipeline = Pipeline([
    ('imputer', SimpleImputer(strategy='most_frequent')),
    ('encoder', OneHotEncoder(sparse_output=False))
])

# 4. 使用 ColumnTransformer 合併兩條管道
preprocessor = ColumnTransformer(
    transformers=[
        ('num', num_pipeline, num_cols),
        ('cat', cat_pipeline, cat_cols)
    ]
)

# 5. 對資料進行 fit_transform
X = df.drop('is_delayed', axis=1)
y = df['is_delayed']
X_transformed = preprocessor.fit_transform(X)

# 6. 印出轉換後的欄位名稱與資料形狀
# 獲取 OneHotEncoder 的欄位名稱
cat_feature_names = preprocessor.named_transformers_['cat'].named_steps['encoder'].get_feature_names_out(cat_cols)
all_feature_names = num_cols + list(cat_feature_names)

print(f"轉換後的欄位名稱：\n{all_feature_names}")
print(f"\n轉換後的資料形狀：{X_transformed.shape}")

# 7. 印出轉換後的前 5 筆資料
print("\n轉換後的前 5 筆資料：")
import pandas as pd
print(pd.DataFrame(X_transformed, columns=all_feature_names).head())
